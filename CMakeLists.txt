# MSVC and clang-cl client, module, and dedicated-server build, plus a native
# GCC/Clang dedicated-server build on Linux. Makefile targets remain
# authoritative on other toolchains; source ownership comes from their shared
# build-mk manifests.
#
# See docs/client-building.md for audio backend and architecture selection, and
# docs/server-building.md for the dedicated server. The Windows dedicated server
# uses cmake/msvc-posix/ for the POSIX names MinGW provides.
#
# Windows:
#   cmake --preset windows-msvc-32bit
#   cmake --build --preset windows-msvc-32bit
#   The 32-bit preset is the recommended Windows build; windows-msvc-64bit
#   explicitly selects the 64-bit port and windows-clangcl-64bit selects the
#   Clang toolset.
#
# Linux:
#   cmake --preset linux-gcc-64bit
#   cmake --build --preset linux-gcc-64bit
#   GCC is the recommended Linux compiler. linux-clang-64bit is the
#   alternative; it selects the acknowledged relaxed floating-point policy
#   because Clang cannot emit the faithful x87 model on x86-64.
cmake_minimum_required(VERSION 3.24)

# Keep generated sources, downloads, and outputs below the root workbench.
cmake_path(IS_PREFIX CMAKE_SOURCE_DIR "${CMAKE_BINARY_DIR}" NORMALIZE _in_source)
set(_workbench "${CMAKE_SOURCE_DIR}/.workbench")
cmake_path(IS_PREFIX _workbench "${CMAKE_BINARY_DIR}" NORMALIZE _in_workbench)
if(_in_source AND NOT _in_workbench)
    message(FATAL_ERROR "Use -B .workbench/build/<name> for an in-repository build.")
endif()

# A native configuration needs a compiler and, for the Makefile generators,
# make. Installing system packages needs root, so report the package that
# provides them instead of letting project() fail with a bare
# compiler-not-found error.
if(NOT CMAKE_HOST_SYSTEM_NAME STREQUAL "Windows")
    find_program(CODUOMP_HOST_CC NAMES cc gcc clang)
    find_program(CODUOMP_HOST_CXX NAMES c++ g++ clang++)
    if(CMAKE_GENERATOR MATCHES "Makefiles")
        find_program(CODUOMP_HOST_MAKE NAMES make gmake)
    endif()
    if(NOT CODUOMP_HOST_CC OR NOT CODUOMP_HOST_CXX OR
        (CMAKE_GENERATOR MATCHES "Makefiles" AND NOT CODUOMP_HOST_MAKE))
        message(FATAL_ERROR
            "CoDUOMP: a C/C++ compiler and make are required for the native "
            "build. Install them with 'sudo apt-get install -y build-essential' "
            "(Debian/Ubuntu) and re-run CMake.")
    endif()
    unset(CODUOMP_HOST_CC CACHE)
    unset(CODUOMP_HOST_CXX CACHE)
    unset(CODUOMP_HOST_MAKE CACHE)
endif()

if(NOT DEFINED CMAKE_BUILD_TYPE OR CMAKE_BUILD_TYPE STREQUAL "")
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Single-configuration build type" FORCE)
endif()
project(CoDUOMP LANGUAGES C CXX)
if(WIN32)
    enable_language(RC)
endif()
set_property(GLOBAL PROPERTY USE_FOLDERS ON)

# The native Linux configuration builds the dedicated server; the client and
# its Windows modules are generated by the Visual Studio configuration.
if(MSVC)
    set(CODUOMP_CLIENT_DEFAULT ON)
    set(CODUOMP_MODULES_DEFAULT ON)
else()
    set(CODUOMP_CLIENT_DEFAULT OFF)
    set(CODUOMP_MODULES_DEFAULT OFF)
endif()

option(CODUOMP_BUILD_CLIENT "Build the client executable" ${CODUOMP_CLIENT_DEFAULT})
option(CODUOMP_BUILD_MODULES "Build cgame, UI, and Windows game modules" ${CODUOMP_MODULES_DEFAULT})
option(CODUOMP_BUILD_SERVER "Build the dedicated server executable" ON)
option(CODUOMP_STATIC_VCRT "Link the MSVC runtime statically" ON)
option(BUNDLED_LIBS "Build pinned third-party dependencies" ON)
option(CODUOMP_CURL_STATIC "Link a static system libcurl (bundled curl is always static)" ON)
option(CODUOMP_RENDERER_GPU_PROFILE "Compile renderer GPU timing" OFF)
option(CODUOMP_DISABLE_SERVER_AUTH "Build the server without external CD-key authorization" ON)
set(CODUOMP_DEP_PREFIX "" CACHE PATH "Prefix containing MSVC or native dependencies")

include(cmake/CoduompManifest.cmake)
include(cmake/CoduompCommon.cmake)
include(cmake/CoduompPlatform.cmake)
include(cmake/CoduompBundledLibs.cmake)
include(cmake/CoduompDependencies.cmake)
include(cmake/CoduompAudio.cmake)
include(cmake/CoduompSources.cmake)
include(cmake/CoduompTargets.cmake)
message(STATUS "${CMAKE_C_COMPILER_ID} ${CODUOMP_ARCH}; ${CODUOMP_FLOAT_MODEL}")
message(STATUS "Output: ${CODUOMP_BIN_ROOT}/<config>")
